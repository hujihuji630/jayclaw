# Web UI v1.1 Warm Document Redesign — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the cyberpunk/neon Web UI with a warm, restrained documentary visual system (moss-green accent, Fraunces/Inter/JetBrains Mono, full-width message blocks, 72ch reading width) and add ⌘K palette, markdown/highlight upgrade, copy/resend/edit, drag-drop upload, keyboard shortcuts, slash commands, and Markdown export.

**Architecture:** Pure-static frontend (no build chain). Vendor scripts and fonts self-hosted under `static/vendor/` and `static/fonts/`. JS becomes ES Modules under `static/modules/`; CSS split via `@import` under `static/styles/`. Backend gets minimal additions: message `id` field on the existing `Message` dataclass (lives in `packages/jay-agent-core/src/jay_agent_core/memory.py`, **not** `agent.py` — the spec had this wrong) and on the web-ui `ChatMessage` Pydantic model; a new `routes/messages.py` with a truncate endpoint; SSE additions for `message_start` / `message_end`; an `/api/sessions/{name}/export.md` route; incremental tokenization in `ContextManager`.

**Tech Stack:** FastAPI + Jinja2 + uvicorn (existing). Vendor JS: `marked` v12, `DOMPurify` v3, `highlight.js` v11, `fuzzysort` v3, `lucide` v0 (icons-only subset). Fonts: Fraunces, Inter, JetBrains Mono (self-hosted woff2). pytest for backend tests.

```html
    <link rel="preconnect" href="https://fonts.googleapis.com">
    <link rel="preconnect" href="https://fonts.gstatic.com" crossorigin>
    <link href="https://fonts.googleapis.com/css2?family=Space+Mono:wght@400;700&family=Syne:wght@400;600;700;800&display=swap" rel="stylesheet">
    <link rel="stylesheet" href="/static/style.css">
```

Replace with:

```html
    <link rel="stylesheet" href="/static/style.css">
    <script>
      // Theme bootstrap — runs before stylesheet evaluates to avoid FOUC.
      (function () {
        try {
          var stored = localStorage.getItem('jayclaw-theme');
          var prefersDark = window.matchMedia && window.matchMedia('(prefers-color-scheme: dark)').matches;
          var theme = stored || (prefersDark ? 'dark' : 'light');
          document.documentElement.setAttribute('data-theme', theme);
        } catch (_) {
          document.documentElement.setAttribute('data-theme', 'light');
        }
      })();
    </script>
```

```python
"""Message-level mutation endpoints (truncate, future: pin, archive)."""

from __future__ import annotations

from fastapi import HTTPException
from pydantic import BaseModel


class TruncateRequest(BaseModel):
    after_id: str


def register(server) -> None:
    """Mount message endpoints onto ``server.app``."""

    @server.app.post("/api/messages/truncate")
    async def truncate(req: TruncateRequest):
        """Remove every message strictly after the message with ``after_id``."""
        target = req.after_id
        idx = next((i for i, m in enumerate(server.history) if m.id == target), None)
        if idx is None:
            raise HTTPException(status_code=400, detail=f"unknown message id: {target}")
        keep = server.history[: idx + 1]
        removed = len(server.history) - len(keep)
        server.history[:] = keep

        # Mirror into core agent's history if present
        core = getattr(server.agent, "agent", None) if server.agent else None
        if core is not None and getattr(core, "history", None) is not None:
            # Best-effort by index — core history may have system/tool messages
            # interleaved, so we truncate by message id when available, else by
            # role-aware index alignment.
            core_hist = core.history
            if all(getattr(m, "id", None) for m in core_hist):
                cidx = next((i for i, m in enumerate(core_hist) if m.id == target), None)
                if cidx is not None:
                    del core_hist[cidx + 1 :]
            else:
                # Fall back to count-by-role
                want_user = sum(1 for m in keep if m.role == 'user')
                want_asst = sum(1 for m in keep if m.role == 'assistant')
                seen_user = seen_asst = 0
                cut_at = len(core_hist)
                for i, m in enumerate(core_hist):
                    if m.role == 'user': seen_user += 1
                    elif m.role == 'assistant': seen_asst += 1
                    if seen_user >= want_user and seen_asst >= want_asst:
                        cut_at = i + 1
                        break
                del core_hist[cut_at:]

        # Persist session truncation
        if server.agent and hasattr(server.agent, "session"):
            try:
                server.agent.session.save()
            except Exception:
                pass

        return {"removed": removed}
```

    async copyMessage(messageDiv) {
        const raw = messageDiv.dataset.raw || '';
        try {
            await navigator.clipboard.writeText(raw);
            const btn = messageDiv.querySelector('[data-action="copy"]');
            const orig = btn.textContent;
            btn.textContent = '✓';
            setTimeout(() => { btn.textContent = orig; }, 1500);
        } catch (_) { /* clipboard denied */ }
    }

    async resendMessage(messageDiv) {
        const id = messageDiv.dataset.messageId;
        const raw = messageDiv.dataset.raw || '';
        if (!id) return;
        // Truncate everything after this message's *predecessor* (we want to
        // remove this user message and everything after, then re-fill input).
        const allMessages = Array.from(this.chatContainer.querySelectorAll('.message'));
        const idx = allMessages.indexOf(messageDiv);
        const prev = idx > 0 ? allMessages[idx - 1] : null;
        const truncateAfter = prev?.dataset.messageId || null;
        if (truncateAfter) {
            try {
                const r = await fetch('/api/messages/truncate', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ after_id: truncateAfter }),
                });
                if (!r.ok) { this.showToast('截断失败', 'error'); return; }
            } catch (e) { this.showToast('截断失败', 'error'); return; }
        } else {
            // No predecessor — clear all history.
            await fetch('/api/history', { method: 'DELETE' });
        }
        // Remove DOM messages from idx onward.
        for (let i = idx; i < allMessages.length; i++) allMessages[i].remove();
        // Refill input.
        this.messageInput.value = raw;
        this.messageInput.dispatchEvent(new Event('input'));
        this.messageInput.focus();
        this.messageInput.selectionStart = this.messageInput.selectionEnd = raw.length;
        this._refreshEditButtons();
    }

    async editMessage(messageDiv) {
        // Same as resend; the spec treats Edit as Resend with cursor positioning.
        return this.resendMessage(messageDiv);
    }

        const val = this.messageInput.value;

    _renderSlash() {
        this.slashPopover.innerHTML = this._slashItems.map((it, i) => `

                <span class="slash-item-cmd">${it.cmd}</span>
                <span class="slash-item-desc">${it.desc}</span>

class ContextManager:
    # ... existing __init__ adds:
    #   self._tok_cache: dict[str, int] = {}    # id -> token count
    #   self._cache_hits: int = 0
    #   self._tokenize_calls: int = 0

    def compute_utilization(
        self,
        messages: list[dict[str, Any]],
        max_tokens: int,
        config: 'CompressionConfig | None' = None,
        previous_ratio: float | None = None,
        model: str | None = None,
    ) -> 'ContextUtilization':
        from .token_counter import count_tokens
        import json as _json

        if not hasattr(self, '_tok_cache'):
            self._tok_cache = {}
            self._cache_hits = 0
            self._tokenize_calls = 0

        total = 0
        for msg in messages:
            mid = msg.get('id')
            cache_key = mid if mid else None
            if cache_key and cache_key in self._tok_cache:
                total += self._tok_cache[cache_key]
                self._cache_hits += 1
                continue
            n = 0
            content = msg.get('content') or ''
            if isinstance(content, str):
                n += count_tokens(content, model=model)
            metadata = msg.get('metadata') or {}
            tool_calls = msg.get('tool_calls') or metadata.get('tool_calls')
            if tool_calls:
                try:
                    n += count_tokens(_json.dumps(tool_calls, ensure_ascii=False), model=model)
                except (TypeError, ValueError):
                    pass
            tool_name = msg.get('name') or metadata.get('name')
            if isinstance(tool_name, str) and tool_name:
                n += count_tokens(tool_name, model=model)
            self._tokenize_calls += 1
            if cache_key:
                self._tok_cache[cache_key] = n
            total += n

        ratio = total / max_tokens if max_tokens > 0 else 0.0
        if config is None:
            config = CompressionConfig()
        if ratio < config.user_decision_threshold:
            zone = 'smart'
        elif ratio < config.level1_threshold:
            zone = 'warning'
        else:
            zone = 'compressed'
        should_prompt = (
            previous_ratio is not None
            and previous_ratio < config.user_decision_threshold
            and ratio >= config.user_decision_threshold
        )
        return ContextUtilization(
            current_tokens=total,
            max_tokens=max_tokens,
            ratio=ratio,
            zone=zone,
            should_prompt_user=should_prompt,
        )

    def invalidate_token_cache(self):
        self._tok_cache = {}
        self._cache_hits = 0
        self._tokenize_calls = 0
